    const strip = `<div class="bgxC-striphead"><b>${t.found}</b><span class="bgx-muted">${list.length - (flying ? 1 : 0)} ${t.wordsN}</span></div><div class="bgxC-strip">${list.map((w, k) =>
      `<div class="bgxC-ticket${k === 0 && (flying || landed) ? (flying ? ' is-new is-hold' : ' is-new') : ''}"${k === 0 && flying ? ' data-bgx="to"' : ''}><span dir="auto">${esc(w)}</span><small class="metric">+${PTS(w)}</small></div>`).join('')}</div>`;
    return `<div class="bgx bgxC${side ? ' is-side' : ''}">${side ? `${arena}<div class="bgxC-side">${top}${word}${strip}</div>` : `${top}${arena}${strip}`}</div>`;
  }
  function tvC(o) {
    const { lang } = o, t = T[lang], P = PLAYERS[lang];
    const pos = [['8%', '10%'], ['8%', 'auto', '10%'], ['auto', '10%', null, '6%'], ['auto', 'auto', '10%', '6%']];
    const seat = (p, k) => {
      const [top, start, end, bottom] = pos[k];
      const s = `${top !== 'auto' ? `top:${top};` : ''}${bottom ? `bottom:${bottom};` : ''}${start !== 'auto' ? `inset-inline-start:${start};` : ''}${end ? `inset-inline-end:${end};` : ''}`;
      return `<div class="bgxC-seat" style="${s}"><div style="position:relative"><div class="bgx-face">${face(p)}</div><span class="bgxC-seat__n metric">${COUNTS[k]}</span></div><b>${esc(p.n)}</b><span class="bgxC-seat__dots">${t.looking}</span></div>`;
    };
    return `<div class="bgx-tv bgxC bgxC-tv" dir="${lang === 'en' ? 'ltr' : 'rtl'}">${tvTop(lang).replace(/<div class="bgx-tv__clock">[^<]*<\/div>/, '')}
      <div class="bgxC-tv__body">${P.map(seat).join('')}<div class="bgxC-arena"><span class="bgxC-clock metric">1:12</span><div class="bgxC-arena__in">${gridHtml(lang, [])}</div></div></div></div>`;
  }
  function endC(o) {
    const { lang, reveal } = o, t = T[lang], P = PLAYERS[lang], cnt = counts(lang), unk = UNKNOWN[lang], p = P[0];
    const pts = roundPts(lang, p, true) - (!reveal ? 5 : 0);
    const tiles = p.words.map((w) => {
      if (w === unk) return `<div class="bgxC-wt is-u"><span dir="auto">${esc(w)}</span><small>${t.yes} <span class="metric">2/3</span> · <span class="metric">+${PTS(w)}</span></small></div>`;
      if (cnt[w] > 1) return `<div class="bgxC-wt is-x"><span dir="auto">${esc(w)}</span><small>${t.shared} <span class="metric">×${cnt[w]}</span></small></div>`;
      if (!reveal) return `<div class="bgxC-wt is-down"><span>${lang === 'en' ? '?' : '؟'}</span><small>${tvLetters(lang, w)}</small></div>`;
      return `<div class="bgxC-wt is-u"><span dir="auto">${esc(w)}</span><small><span class="metric">+${PTS(w)}</span> · ${t.unique}</small></div>`;
    }).join('');
    const minis = P.map((q, k) => `<div class="bgxC-mini${k === 0 ? ' is-on' : ''}"><div class="bgx-face">${face(q)}</div>${esc(q.n)}</div>`).join('');
    const body = `<div class="bgx bgxC-res"><div class="bgx-row"><span class="bgx-h bgx-grow">${t.result}</span><span class="bgx-muted">1 / 4</span></div><div class="bgxC-faces">${minis}</div>
      <div class="bgxC-spot"><div class="bgx-face">${face(p)}</div><div class="bgxC-spot__name">${esc(p.n)}</div><div class="bgxC-tiles">${tiles}</div><div class="bgxC-total metric">+${pts}</div></div>
      <button class="btn btn--primary btn--lg">${lang === 'en' ? 'Next player' : 'اللي بعده'}</button></div>`;
    if (!o.vote) return body;
    const tr = { ar: [10, 5, 6, 3], en: [4, 9, 5, 6] }[lang];
    return body + `<div class="bgxC-dim"><div class="bgxC-dialog"><div class="bgx-row" style="justify-content:center"><div class="bgx-face" style="width:2rem;height:2rem">${face(p)}</div><span class="bgx-muted">${esc(p.n)} ${t.wrote}</span></div>
      <div class="bgxC-dialog__word" dir="auto">${esc(unk)}</div><div class="bgxC-trace bgxC">${gridHtml(lang, tr)}</div><span class="bgx-muted">${t.vote}</span>
      <button class="btn btn--primary btn--lg">✓ ${t.yes}</button><button class="btn btn--ghost">✕ ${t.no}</button><span class="bgx-muted"><span class="metric">2 / 4</span> ${t.votes}</span></div></div>`;
  }
  const tvLetters = (lang, w) => lang === 'en' ? `${w.length} letters` : `${w.length} حروف`;

  window.bgx = function (o) {
    document.querySelectorAll('.bgx-ghost, .bgx-tv, .bgxC-dim').forEach((n) => n.remove());
    const fn = { A: { play: playA, tv: tvA, end: endA }, B: { play: playB, tv: tvB, end: endB }, C: { play: playC, tv: tvC, end: endC } }[o.look][o.screen];
    const accent = { A: 'amber', B: 'blue', C: 'violet' }[o.look];
    const html = fn(o);
    let root;
    if (o.screen === 'tv') {
      document.body.insertAdjacentHTML('beforeend', html);
      root = document.querySelector('.bgx-tv');
      root.setAttribute('data-accent', accent);
    } else {
      const view = document.getElementById('view-play-wordwheel');
      view.setAttribute('data-accent', accent);
      view.innerHTML = `<div style="position:relative">${html}</div>`;
      const dim = view.querySelector('.bgxC-dim');
      if (dim) { dim.setAttribute('data-accent', accent); document.body.appendChild(dim); }
      root = view;
    }
    const tr = TRACE[o.lang];
    if (o.screen === 'play' && o.trace && !(o.fly > 0)) drawLine(root, tr.p.slice(0, o.trace), true);
    if (o.screen === 'end' && o.look === 'C' && o.vote) drawLine(document.querySelector('.bgxC-dim'), { ar: [10, 5, 6, 3], en: [4, 9, 5, 6] }[o.lang], false);
    if (o.fly > 0 && o.fly < 1) flyGhost(root, o.fly);
    return 1;
  };
})();
