const card = document.querySelector('.hm-write-card--local');
const ic = card.querySelector('.pass-poster__icon').innerHTML;
const name = card.querySelector('.pass-poster__name').textContent;
const label = card.querySelector('.pass-poster__label').textContent;
const away = card.querySelector('.hm-write__away').textContent;
const ar = document.documentElement.lang === 'ar';
// B: a slim title in the card and the warning as a banner above it
card.insertAdjacentHTML('afterbegin', `<div class="hm-b-title"><span class="hm-b-ic">${ic}</span>${label}</div>`);
card.insertAdjacentHTML('beforebegin', `<div class="hm-b-banner">${away}</div>`);
// C: «أنا ...، يلا» under the poster, and a small head over the form
card.querySelector('.hm-write__away').insertAdjacentHTML('afterend', `<button type="button" class="btn btn--primary btn--lg hm-c-go">${ar ? 'أنا ' + name + '، يلا ✍️' : "I'm " + name + ', go ✍️'}</button>`);
card.insertAdjacentHTML('afterbegin', `<div class="hm-c-head"><b>✍️ ${name}</b><span>${ar ? 'محدش يبص' : 'no peeking'}</span></div>`);
if (window.__V === 'C') document.body.classList.add(window.__STEP || 'c2');
