  /* ---------------- look C ---------------- */
  function lookC(L, st) {
    const n = st === 'play' ? 7 : st === 'lose' ? 20 : 13;
    const bar = (sub, glow, typing) => `<div class="orxC-bar"><div class="orxC-av ${glow ? 'is-glow' : ''}">${oracle({ mood: glow ? 'think' : 'idle', ball: 'q', head: true })}</div><div class="orxC-bar__txt"><span class="orxC-bar__name">${L.title}</span><span class="orxC-bar__sub ${typing ? 'is-typing' : ''}">${sub}</span><div class="orxC-prog"><i style="width:${n / 20 * 100}%"></i></div></div>${guessesHtml(L, st === 'lose' ? 0 : 3)}</div>`;
    const sub = `<span>${L.qOf(n, 20).replace(/<[^>]+>/g, '')}</span>`;
    const o = (t, nn, q) => `<div class="orxC-msg orxC-msg--o ${q ? 'is-q' : ''}">${nn ? `<span class="orxC-n">${L.qOf(nn, 20).replace(/<[^>]+>/g, '')}</span>` : ''}${t}</div>`;
    const me = (t) => `<div class="orxC-msg orxC-msg--me">${t}</div>`;
    const hist = (upto) => { let s = ''; for (let i = 0; i < upto; i++) s += o(L.qs[i]) + me(L[L.ans[i]]); return s; };
    const dockPlay = `<div class="orxC-dock"><div class="orx-pair"><button class="btn btn--success btn--lg">${L.y}</button><button class="btn btn--danger btn--lg">${L.n}</button></div><div class="orxC-chips"><button class="btn btn--secondary btn--sm">${L.py}</button><button class="btn btn--neutral btn--sm">${L.dk}</button><button class="btn btn--secondary btn--sm">${L.pn}</button></div></div>`;
    const dockGuess = `<div class="orxC-dock"><div class="orx-pair"><button class="btn btn--success btn--lg">${L.yes.replace(' 🎉', '')}</button><button class="btn btn--ghost btn--lg">${L.no}</button></div><div class="orx-note">${L.leftAfter}</div></div>`;
    const card = (style = '') => `<div class="orxC-card" style="${style}"><small>${L.guessCard}</small>${silhouette('⚽', 'orx-sil--big')}<div class="orx-name">${L.name}؟</div><div class="orx-kind">${L.kind}</div></div>`;
    const portrait = (mood, ball, line) => `<div class="orxC-portrait">${oracle({ mood, ball })}<div class="orx-name" style="font-size:var(--fs-h2)">${L.title}</div><div class="orx-note">${line || L.online}</div></div>`;
    const wrap = (p, b, log, dock) => `<div class="orx orxC">${p}<div class="orxC-main">${b}<div class="orxC-log">${log}</div>${dock}</div></div>`;
    const late = o(L.qs[4]) + me(L.y) + o(L.qs[5]) + me(L.py) + o(L.qs[6]) + me(L.y) + o(L.lateQ[0]) + me(L.dk) + o(L.lateQ[1]) + me(L.y);
    if (st === 'play') return wrap(portrait('idle', 'q'), bar(sub), `<span class="orxC-day">${L.today}</span>${hist(6)}<button class="orxC-undo">${L.changed}</button>${o(L.qs[6], 7, true)}`, dockPlay);
    if (st === 'f1') return wrap(portrait('idle', 'q'), bar(sub), `${late.replace(/<div class="orxC-msg orxC-msg--me">[^<]*<\/div>$/, '')}<div class="orxC-msg orxC-msg--me" style="transform:translateY(46px) scale(.9);opacity:.6">${L.y}</div>`, dockPlay);
    if (st === 'f2') return wrap(portrait('think', 'dim', L.thinkC), bar(L.typing, true, true), `${late}<div class="orxC-msg orxC-msg--o"><span class="orx-dots" style="color:var(--accent-ink)"><i></i><i></i><i></i></span></div>`, dockPlay.replace('class="orxC-dock"', 'class="orxC-dock" style="opacity:.35"'));
    if (st === 'f3') return wrap(portrait('think', 'glow', L.thinkC), bar(L.typing, true, true), `${late}${o(L.sure)}<div class="orxC-msg orxC-msg--o"><span class="orx-dots" style="color:var(--accent-ink)"><i></i><i></i><i></i></span></div>`, dockPlay.replace('class="orxC-dock"', 'class="orxC-dock" style="opacity:.35"'));
    if (st === 'f4') return wrap(portrait('happy', 'face'), bar(sub, true), `${late}${o(L.sure)}${card('transform:scale(.7);transform-origin:var(--o, 100%) 100%;opacity:.55')}`, dockGuess.replace('class="orxC-dock"', 'class="orxC-dock" style="opacity:.35"'));
    if (st === 'guess' || st === 'f5') return wrap(portrait('happy', 'face'), bar(sub), `${late}${o(L.sure)}${card()}`, dockGuess);
    if (st === 'win') return wrap(portrait('happy', 'face'), bar(`${L.winC} 13 ${L.inQ}`), `${late}${o(L.sure)}${card('opacity:.85')}${me(L.yes)}<div class="orxC-msg orxC-msg--o" style="font-size:var(--fs-h2);font-weight:900">${L.win}<div class="orx-stat" style="margin-top:8px"><div><span class="metric">13</span><small>${L.inQ}</small></div><div><span class="metric">1</span><small>${L.guessN}</small></div></div></div>`,
      `<div class="orxC-dock"><div class="orx-pair"><button class="btn btn--primary btn--lg">${L.again}</button><button class="btn btn--ghost btn--lg">${L.exit}</button></div></div>`);
    if (st === 'lose') return wrap(portrait('puzzled', 'dim'), bar(L.usedAll), `${late}${o(L.guessCard + ' ' + 'أبو تريكة؟'.replace('أبو تريكة', L === T.en ? 'Aboutrika' : 'أبو تريكة'))}${me(L.no)}<div class="orxC-msg orxC-msg--o" style="font-size:var(--fs-h2);font-weight:900">${L.lose}<br>${L.who}</div><span class="orxC-day">${L.learn}</span>`,
      `<div class="orxC-dock"><div class="orxC-compose"><input placeholder="${L.ph}"><button class="orxC-send">${appState.lang === 'ar' ? '➤' : '➤'}</button></div><div class="orx-pair"><button class="btn btn--secondary btn--sm">${L.again}</button><button class="btn btn--ghost btn--sm">${L.exit}</button></div></div>`);
  }

  window.orxShow = function (look, st) {
    let s = document.getElementById('orx-css');
    if (!s) { s = document.createElement('style'); s.id = 'orx-css'; document.head.appendChild(s); }
    s.textContent = css;
    const L = T[appState.lang] || T.ar;
    const view = document.getElementById('view-play-guessnum');
    view.dataset.accent = { A: 'violet', B: 'teal', C: 'indigo' }[look];
    view.innerHTML = (look === 'A' ? lookA : look === 'B' ? lookB : lookC)(L, st);
    const root = view.querySelector('.orx'), main = document.getElementById('shell-main') || document.querySelector('main');
    if (look === 'C' && root && main) { const mb = main.getBoundingClientRect(), rb = root.getBoundingClientRect(); root.style.height = Math.floor(mb.bottom - rb.top - 12) + 'px'; }
    const tt = document.getElementById('app-title-text'); if (tt) tt.textContent = L.title;
    const ti = document.getElementById('app-title-icon'); if (ti) ti.textContent = '🔮';
    return view.innerHTML.length;
  };
