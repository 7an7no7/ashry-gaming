  function placeGhosts(list) {
    document.querySelectorAll('.hx-ghost').forEach(g => g.remove());
    list.forEach(g => {
      const from = document.querySelector(g.from), to = document.querySelector(g.to);
      if (!from || !to) return;
      const a = from.getBoundingClientRect(), b = to.getBoundingClientRect();
      const ref = g.useTo ? b : a;
      const w = ref.width, h = ref.height;
      const ax = a.left + a.width / 2, ay = a.top + a.height / 2, bx = b.left + b.width / 2, by = b.top + b.height / 2;
      const x = ax + (bx - ax) * g.t, y = ay + (by - ay) * g.t - Math.sin(Math.PI * g.t) * 22;
      let el;
      if (g.html) { el = document.createElement(g.cls && g.cls.includes('hxb-key') ? 'button' : 'span'); el.className = g.cls || ''; el.innerHTML = g.html; }
      else { el = from.cloneNode(true); el.classList.remove('is-lift', 'is-sel', 'is-used', 'is-press'); el.removeAttribute('data-slot'); el.removeAttribute('style'); }
      if (g.from === '[data-slot="0"]' && g.keepSize && from.classList.contains('hxa-tile')) el.className = 'hxa-tile is-new is-l3';
      el.classList.add('hx-ghost'); const av = document.querySelector('[data-accent] .hx'); if (av) el.setAttribute('data-accent', av.parentElement.getAttribute('data-accent'));
      el.style.width = w + 'px'; el.style.height = h + 'px';
      el.style.left = (x - w / 2) + 'px'; el.style.top = (y - h / 2) + 'px';
      el.style.transform = `scale(${g.scale || 1})`;
      el.style.boxShadow = '0 18px 30px -10px rgba(0,0,0,.35)';
      if (g.fade) el.style.opacity = g.fade;
      document.body.appendChild(el);
      // The page's fixed boxes may sit in an offset containing block: correct by what was measured.
      const r = el.getBoundingClientRect();
      const dx = (r.left + r.width / 2) - x, dy = (r.top + r.height / 2) - y;
      el.style.left = (x - w / 2 - dx) + 'px'; el.style.top = (y - h / 2 - dy) + 'px';
    });
  }

  window.hxShow = async function (look, screen, frame) {
    const L0 = LOOKS[look];
    document.querySelectorAll('.hx-ghost').forEach(g => g.remove());
    if (screen === 'tv' || screen === 'tvend') {
      if (appState.currentView !== 'room-tv') { setView('room-tv'); await new Promise(r => setTimeout(r, 300)); }
      const v = document.getElementById('view-room-tv');
      v.classList.remove('hidden');
      v.setAttribute('data-accent', L0.accent);
      v.innerHTML = screen === 'tv' ? L0.tv() : `<div class="hxtv hxtv-end">${L0.end()}</div>`;
    } else {
      if (appState.currentView !== 'play-wordwheel') {
        if (typeof startWordWheel !== 'function') { setView('setup-wordwheel'); await new Promise(r => setTimeout(r, 1200)); }
        startWordWheel(false); await new Promise(r => setTimeout(r, 500));
      }
      if (typeof wheelTimer !== 'undefined' && wheelTimer) wheelTimer.stop();
      const v = document.getElementById('view-play-wordwheel');
      v.setAttribute('data-accent', L0.accent);
      v.innerHTML = screen === 'end' ? L0.end() : L0.play(scene(frame));
      const tt = document.getElementById('app-title-text'); if (tt) tt.textContent = L('حسبة', 'Numbers');
      const ti = document.getElementById('app-title-icon'); if (ti) ti.textContent = '🧮';
      const m = document.querySelector('.shell__main') || document.querySelector('main'); if (m) m.scrollTop = 0;
      await new Promise(r => requestAnimationFrame(() => requestAnimationFrame(r)));
      if (screen === 'play' && frame) placeGhosts(L0.ghosts(frame));
    }
    return 1;
  };
