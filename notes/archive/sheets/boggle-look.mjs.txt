/**
 * شبكة الحروف look sheet: node shoot.mjs [prefix,prefix]  (the preview at http://localhost:4321 must be running)
 * Injects looks.js into the real app (the play-wordwheel view, or a TV stage over it) and shoots each look into shots/.
 */
import fs from 'node:fs';
import path from 'node:path';
import { spawn } from 'node:child_process';
import { fileURLToPath } from 'node:url';

const here = fileURLToPath(new URL('./', import.meta.url));
const out = path.join(here, "shots");
fs.mkdirSync(out, { recursive: true });
const BASE = 'http://localhost:4321/';
const wait = (ms) => new Promise((r) => setTimeout(r, ms));
const chromePath = ['C:/Program Files/Google/Chrome/Application/chrome.exe', 'C:/Program Files (x86)/Google/Chrome/Application/chrome.exe'].find((p) => fs.existsSync(p));
const profile = fs.mkdtempSync(path.join(here, 'chrome-'));
const chrome = spawn(chromePath, ['--headless=new', '--remote-debugging-port=0', '--no-first-run', '--no-default-browser-check', '--user-data-dir=' + profile, 'about:blank'], { stdio: 'ignore' });
let wsUrl = null;
for (let i = 0; i < 100 && !wsUrl; i++) {
  await wait(100);
  try { const [port, p] = fs.readFileSync(path.join(profile, 'DevToolsActivePort'), 'utf8').split('\n'); wsUrl = `ws://127.0.0.1:${port}${p}`; } catch (e) {}
}
const ws = new WebSocket(wsUrl);
await new Promise((r, j) => { ws.onopen = r; ws.onerror = j; });
let msgId = 0;
const pending = new Map();
const errors = [];
ws.onmessage = (e) => {
  const m = JSON.parse(e.data);
  if (m.id && pending.has(m.id)) { pending.get(m.id)(m); pending.delete(m.id); return; }
  if (m.method === 'Runtime.exceptionThrown') errors.push(JSON.stringify(m.params.exceptionDetails.exception && m.params.exceptionDetails.exception.description).slice(0, 200));
};
const send = (method, params = {}, sessionId) => new Promise((resolve, reject) => {
  const id = ++msgId;
  pending.set(id, (m) => (m.error ? reject(new Error(method + ': ' + m.error.message)) : resolve(m.result)));
  ws.send(JSON.stringify({ id, method, params, sessionId }));
});
const { targetId } = await send('Target.createTarget', { url: 'about:blank' });
const { sessionId } = await send('Target.attachToTarget', { targetId, flatten: true });
await send('Runtime.enable', {}, sessionId);
await send('Page.enable', {}, sessionId);
const size = (w, h) => send('Emulation.setDeviceMetricsOverride', { width: w, height: h, deviceScaleFactor: 2, mobile: w < 768 }, sessionId);
async function ev(expr) {
  const r = await send('Runtime.evaluate', { expression: expr, awaitPromise: true, returnByValue: true }, sessionId);
  if (r.exceptionDetails) throw new Error((r.exceptionDetails.exception && r.exceptionDetails.exception.description) || r.exceptionDetails.text);
  return r.result && r.result.value;
}
await size(375, 812);
await send('Page.navigate', { url: BASE }, sessionId);
for (let i = 0; i < 100; i++) {
  await wait(200);
  const ok = await ev(`(() => { if (typeof appState === 'undefined' || document.readyState !== 'complete') return false; const l = document.getElementById('app-loader'); return !l || l.style.display === 'none' || l.classList.contains('hidden'); })()`).catch(() => false);
  if (ok) break;
}
await wait(1500);

const looksJs = fs.readFileSync(path.join(here, 'looks.js'), 'utf8');
await ev(looksJs + '; 1');
await ev(`(async () => { setView('play-wordwheel'); await new Promise(r => setTimeout(r, 1500)); return 1; })()`);
const only = process.argv[2] ? process.argv[2].split(',') : null;
async function shoot(file, o, w, h) {
  if (only && !only.some((x) => file.startsWith(x))) return;
  await size(w, h);
  await ev(`(async () => {
    if (appState.lang !== '${o.lang}') toggleLanguage();
    if (document.body.classList.contains('dark') !== ${!!o.dark}) toggleDarkMode();
    if (appState.currentView !== 'play-wordwheel') { setView('play-wordwheel'); await new Promise(r => setTimeout(r, 800)); }
    let s = document.getElementById('bgx-still'); if (!s) { s = document.createElement('style'); s.id = 'bgx-still'; document.head.appendChild(s); }
    s.textContent = '*{transition:none!important;animation:none!important}';
    document.querySelectorAll('.modal-overlay:not(.hidden)').forEach(m => m.classList.add('hidden'));
    window.dispatchEvent(new Event('resize'));
    if (typeof lzEnsure === 'function') { try { await lzEnsure(['faces']); } catch (e) {} }
    await new Promise(r => setTimeout(r, 300));
    bgx(${JSON.stringify(o)});
    const tt = document.getElementById('app-title-text'); if (tt) tt.textContent = '${o.lang}' === 'en' ? 'Letter Grid' : 'شبكة الحروف'; const ti = document.getElementById('app-title-icon'); if (ti) ti.textContent = '🔠';
    const m = document.getElementById('shell-main') || document.querySelector('main'); if (m) m.scrollTop = 0;
    return 1; })()`);
  await wait(500);
  const { data } = await send('Page.captureScreenshot', { format: 'jpeg', quality: 80 }, sessionId);
  fs.writeFileSync(path.join(out, file + '.jpg'), Buffer.from(data, 'base64'));
  console.log('shot', file);
}
for (const look of ['A', 'B', 'C']) {
  const L = (x) => Object.assign({ look, lang: 'ar', dark: false, screen: 'play', trace: 5, fly: 0 }, x);
  await shoot(`${look}-play-ar-light`, L({}), 375, 812);
  await shoot(`${look}-play-ar-dark`, L({ dark: true }), 375, 812);
  await shoot(`${look}-play-en-dark`, L({ lang: 'en', dark: true, trace: 6 }), 375, 812);
  await shoot(`${look}-side-ar-light`, L({ side: true }), 667, 375);
  await shoot(`${look}-tv-ar`, L({ screen: 'tv', dark: true }), 1280, 720);
  await shoot(`${look}-end-ar-light`, L({ screen: 'end', reveal: 1, vote: look !== 'C' }), 375, 812);
  await shoot(`${look}-end-ar-dark`, L({ screen: 'end', reveal: 1, vote: true, dark: true }), 375, 812);
  await shoot(`${look}-f1-trace`, L({ trace: 2 }), 375, 812);
  await shoot(`${look}-f2-traced`, L({ trace: 5 }), 375, 812);
  await shoot(`${look}-f3-fly`, L({ trace: 5, fly: 0.45 }), 375, 812);
  await shoot(`${look}-f4-landed`, L({ trace: 0, fly: 1 }), 375, 812);
  await shoot(`${look}-f5-reveal`, L({ screen: 'end', reveal: 0, vote: false }), 375, 812);
  await shoot(`${look}-f6-unique`, L({ screen: 'end', reveal: 1, vote: false }), 375, 812);
}
console.log('errors:', errors.length ? errors.join('\n') : 'none');
ws.close();
chrome.kill();
setTimeout(() => { try { fs.rmSync(profile, { recursive: true, force: true }); } catch (e) {} process.exit(0); }, 500);
