      }
    }
    return null;
  }

  function placePop() {
    const pop = document.querySelector('[data-pop]'); const g = document.querySelector('.bw-ghost'); if (!pop || !g) return;
    const host = pop.parentElement.getBoundingClientRect(); const r = g.getBoundingClientRect();
    pop.style.left = (r.left + r.width / 2 - host.left) + 'px'; pop.style.top = (r.bottom - host.top + 12) + 'px';
    if (pop.dataset.anchor === 'start') { pop.style.transform = 'none'; pop.style.left = (r.left - host.left) + 'px'; pop.classList.add('is-start'); }
    const pr = pop.getBoundingClientRect();
    if (pr.left < host.left + 4) pop.style.left = (parseFloat(pop.style.left) + host.left + 4 - pr.left) + 'px';
    if (pr.right > host.right - 4) pop.style.left = (parseFloat(pop.style.left) - (pr.right - host.right + 4)) + 'px';
  }
